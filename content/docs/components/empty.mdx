## Usage

```tsx
import {
  Empty,
  EmptyHeader,
  EmptyMedia,
  EmptyTitle,
  EmptyDescription,
  EmptyContent,
} from "@/components/ui/empty"
```

```tsx
<Empty>
  <EmptyHeader>
    <EmptyMedia variant="icon">
      <FileX2 />
    </EmptyMedia>
    <EmptyTitle>No files found</EmptyTitle>
    <EmptyDescription>
      Upload your first file to get started.
    </EmptyDescription>
  </EmptyHeader>
  <EmptyContent>
    <Button>Upload File</Button>
  </EmptyContent>
</Empty>
```
